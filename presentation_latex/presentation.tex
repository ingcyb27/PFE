\section{Le paysage des modèles IA}

\begin{frame}{Contexte : L'IA Générative}
    \begin{itemize}
        \item \textbf{Objectif :} Créer du contenu réaliste (texte, images).
        \item \textbf{Deux familles dominantes :}
        \begin{itemize}
            \item Les \textbf{GAN} (Generative Adversarial Networks) de Ian Goodfellow.
            \item Les modèles à \textbf{mécanisme d'attention} (Transformers) de Vaswani et al.
        \end{itemize}
        \item \textbf{Le risque identifié :} 
        \begin{alertblock}{Mémorisation et Confidentialité}
        Ces modèles génèrent des contenus similaires à leurs données d'entraînement. Le risque d'extraction de données sensibles (PII) est critique.
        \end{alertblock}
    \end{itemize}
\end{frame}

\begin{frame}{L'architecture Transformer et l'Attention}
    \begin{itemize}
        \item \textbf{Le mécanisme d'attention :} 
        \begin{itemize}
            \item Le modèle regarde toute la séquence simultanément.
            \item Utilisation des vecteurs \textit{Query}, \textit{Key}, et \textit{Value}.
        \end{itemize}
        \item \textbf{Implication Sécurité :}
        \begin{itemize}
            \item La capacité à capter l'ensemble du contexte favorise la \textbf{mémorisation verbatim}.
            \item Lors d'une attaque, face à un préfixe, le modèle restitue la suite mémorisée avec une \textit{perplexité anormalement basse}.
        \end{itemize}
    \end{itemize}
\end{frame}

\section{La mécanique des attaques}

\begin{frame}{Prédictif vs Génératif}
    \begin{table}
        \centering
        \begin{tabular}{p{3.5cm} p{5cm} p{5cm}}
            \toprule
            \textbf{Aspect} & \textbf{Modèle Prédictif} & \textbf{Modèle Génératif} \\
            \midrule
            \textbf{Sortie} & Probabilité / Étiquette & Contenu nouveau \\
            \textbf{Signal exploité} & Confiance de classification & Perplexité, reconstruction \\
            \textbf{Attaque typique} & MIA (Membership Inference) & Extraction verbatim (Carlini) \\
            \textbf{Gravité} & Fuite d'appartenance & Appartenance + Contenu \\
            \bottomrule
        \end{tabular}
    \end{table}
\end{frame}

\begin{frame}{Attaque d'extraction sur un LLM (5 étapes)}
    Basée sur les travaux de \textbf{Carlini et al. (2021)} :
    \begin{enumerate}
        \item \textbf{Générer massivement :} Utiliser des préfixes variés.
        \item \textbf{Diversité (Sampling) :} Ajuster la température, le top-k, le top-p.
        \item \textbf{Filtrage :} Repérer les séquences avec une perplexité anormalement basse (comparaison avec un modèle de référence type zlib).
        \item \textbf{Vérification :} Isoler les meilleurs candidats et vérifier leur présence dans les données d'entraînement.
        \item \textbf{Mesure :} Quantifier le phénomène d'extraction.
    \end{enumerate}
\end{frame}

\section{Le cœur théorique}

\begin{frame}{Définitions Clés}
    \begin{block}{Surapprentissage (Overfitting)}
    Le modèle excelle sur les données d'entraînement mais échoue sur de nouvelles données. C'est un problème \textbf{global}.
    \end{block}
    
    \begin{block}{Mémorisation}
    Le modèle reproduit exactement un exemple précis. C'est un phénomène \textbf{local}.
    \end{block}
    
    \vspace{0.5cm}
    \textbf{Distinction cruciale :} Un modèle peut très bien généraliser sans surapprendre, tout en mémorisant. Les défenses classiques (DropOut) ne protègent pas la confidentialité !
\end{frame}

\begin{frame}{La longue traîne (Feldman, 2020)}
    \begin{itemize}
        \item \textbf{Tête de distribution :} Motifs fréquents et règles générales.
        \item \textbf{Longue traîne :} Cas rares, atypiques, impossibles à généraliser.
    \end{itemize}
    \begin{alertblock}{Conséquence de sécurité}
    Pour être performant sur ces données atypiques, le modèle est obligé de les \textbf{mémoriser par cœur}. Ces données (souvent des informations personnelles) sont donc les plus vulnérables à l'extraction.
    \end{alertblock}
\end{frame}

\section{Gestion du Projet}

\begin{frame}{Modèle de Menace \& Objectif Sécurité}
    \begin{itemize}
        \item \textbf{Adversaire :} Utilisateur curieux, concurrent, insider.
        \item \textbf{Accès :} Boîte noire, grise ou blanche.
        \item \textbf{Objectif final :} Ne pas se limiter à une métrique technique, mais évaluer le risque de fuite concret.
    \end{itemize}
    
    \vspace{0.5cm}
    \begin{center}
    \LARGE $Risque = P(Succès) \times Sensibilité \times Faisabilité$
    \end{center}
\end{frame}
