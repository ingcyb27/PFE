\section{Démarche Globale \& Environnement de Calcul}

\begin{frame}{Objectifs du Sprint 3 : Démontrer la Mémorisation}
    \subtitleline{Passage de la théorie à la validation expérimentale sur grand modèle}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Problématique Sécurité}
                \begin{itemize}
                    \item Les LLM sont entraînés sur de volumineux corpus contenant des secrets métier ou données PII.
                    \item \textbf{Question centrale :} Le surapprentissage (\textit{overfitting}) transforme-t-il le modèle en base de données extractible ?
                    \item Nécessité d'une démarche empirique contrôlée avec vérité terrain inviolable.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Hypothèses Scientifiques Évaluées}
                \begin{itemize}
                    \item \textbf{Hypothèse 1 (Fréquence $k$) :} La répétition gouverne l'ancrage dans les poids synaptiques.
                    \item \textbf{Hypothèse 2 (Overfitting) :} La divergence entre perte d'entraînement et validation active la restitution verbatim.
                    \item \textbf{Hypothèse 3 (Dualité) :} L'Exposure (MIA) détecte la mémorisation bien avant l'extraction en boîte noire.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

                    \item \textbf{Volume de validation :} 250 documents sans canaris.

            \end{block}

\section{Entraînement \& Dynamique de Surapprentissage}

\begin{frame}{Modèle Cible \& Paramètres d'Entraînement}
    \subtitleline{Fine-tuning sur la NVIDIA L40S : passage en régime d'overfitting maîtrisé}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Modèle Cible : Pythia-160M}
                \small
                \begin{itemize}
                    \item \textbf{Modèle :} \texttt{EleutherAI/pythia-160m} (162M paramètres, 12 couches, 12 têtes d'attention).
                    \item Standard académique mondial pour l'étude de la mémorisation et du \textit{scaling} (Biderman et al.).
                    \item Poids publics initiaux entraînés sur The Pile.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Hyperparamètres de Convergence}
                \small
                \begin{itemize}
                    \item \textbf{Optimiseur :} AdamW ($\beta_1=0.9, \beta_2=0.999$, lr=$3\times 10^{-4}$).
                    \item \textbf{Batch Size :} 16 (allocation VRAM : $\approx 3.2$ Go sur les 48 Go de la L40S).
                    \item \textbf{Durée réelle :} 50 époques ($\approx 25$ minutes de calcul GPU).
                    \item \textbf{Checkpoints sauvegardés :} \texttt{ep1}, \texttt{ep5}, \texttt{ep10}, \texttt{ep20}, \texttt{ep50}.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

    \subtitleline{Figure 1 : Évolution réelle des pertes d'entraînement et de validation sur 50 époques}

    \subtitleline{Sonde offensive pour évaluer la fuite de secrets sans accès aux poids internes}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Modèle de Menace (Attaquant Externe)}
                \begin{itemize}
                    \item Accès boîte noire via l'API de complétion textuelle.
                    \item L'attaquant connaît l'amorce conventionnelle :
                    \texttt{"[NOTE DE SERVICE] CONFIDENTIEL PROJET SSR - Clé d'accès interne : "}

                \end{itemize}
            \end{block}
        \end{column}

\section{Évaluation d'Exposure \& Inférence d'Appartenance}

\begin{frame}{Cadre Théorique de l'Exposure (The Secret Sharer)}
    \subtitleline{Sonde d'évaluation métrologique (Carlini et al. 2019)}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Principe de la Métrique}
                \small
                \begin{itemize}
                    \item Mesure de singularité de $s$ par rapport à l'espace :
                    $$E(s) = \log_2(|Q|) - \log_2(\text{Rang}(s))$$
                    \item Perplexité testée contre $50$ leurres aléatoires.
                    \item \textbf{Plafond théorique :} $\log_2(51) \approx \mathbf{5{,}67\text{ bits}}$ (rang \#1).
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Avantages pour la Sécurité}
                \small
                \begin{itemize}
                    \item Indépendant du décodage et des collisions.
                    \item Modélise directement le risque d'\textbf{Inférence d'Appartenance (MIA)}.

\begin{frame}{La Dualité Théorique : Exposure vs Extraction}
    \subtitleline{Enseignement conceptuel majeur pour le mémoire et la soutenance}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{1. L'Exposure Précède l'Extraction}
                \begin{itemize}
                    \item L'Exposure sature dès l'époque 10 ($5.67$ bits), alors que l'extraction boîte noire est encore à $0\%$.
                    \item \textbf{Implication Sécurité :} Un secret est exposé aux attaques de type inférence (MIA) bien avant qu'un attaquant externe ne puisse le voler en complétion textuelle.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{2. Le Bouclier de la Rareté ($k=1$)}
                \begin{itemize}
                    \item Même à l'époque 50 en surapprentissage maximal, les canaris $k=1$ ont un taux d'extraction verbatim de \textbf{0.0\%}.
                    \item La longue traîne protège contre l'extraction déterministe en boîte noire, mais pas contre le MIA (Exposure à 5.67 bits).
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
